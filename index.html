function parseOzonCSV(text) { ozonData = []; // Нормализуем переносы строк (и \r\n, и \n) const lines = text.split(/\r?\n/); if (lines.length < 2) { console.error('Файл пуст или содержит только заголовки'); return; } const firstLine = lines; // Определяем разделитель: если есть ';', считаем, что это он (стандарт Ozon) const delimiter = firstLine.includes(';') ? ';' : ','; // Парсим заголовки с учетом кавычек const headers = parseCSVLine(firstLine, delimiter).map(h => h.toLowerCase().trim()); // Маппинг колонок: ищем по частичному совпадению, чтобы не зависеть от точных формулировок Ozon const colMap = { date: findColumnIndex(headers, ['дата отгрузки', 'дата заказа']), sku: findColumnIndex(headers, ['sku', 'артикул']), name: findColumnIndex(headers, ['название товара', 'наименование']), pricePaid: findColumnIndex(headers, ['оплачено покупателем', 'сумма к оплате']), priceYour: findColumnIndex(headers, ['ваша цена', 'цена продажи']), logistics: findColumnIndex(headers, ['стоимость доставки', 'логистика']), quantity: findColumnIndex(headers, ['количество']) }; // Проверка на наличие обязательных колонок if (colMap.date === -1 || colMap.sku === -1 || colMap.pricePaid === -1) { console.error('Не найдены обязательные колонки. Доступные заголовки:', headers); alert('Ошибка: в файле не найдены обязательные колонки (Дата отгрузки, SKU, Оплачено покупателем). Проверьте выгрузку из ЛК Ozon.'); return; } for (let i = 1; i < lines.length; i++) { const line = lines[i].trim(); if (!line) continue; const parts = parseCSVLine(line, delimiter); // Пропускаем строки, где меньше колонок, чем нужно if (parts.length <= colMap.pricePaid) continue; const cleanNumber = (val) => { if (typeof val !== 'string') return 0; // Удаляем кавычки, пробелы, заменяем запятую на точку const cleaned = val.replace(/"/g, '').replace(/\s+/g, '').replace(',', '.'); const num = parseFloat(cleaned); return isNaN(num) ? 0 : num; }; const yourPrice = cleanNumber(parts[colMap.priceYour]); const paidPrice = cleanNumber(parts[colMap.pricePaid]); const commission = yourPrice - paidPrice; // Комиссия Ozon ozonData.push({ date: parts[colMap.date] ? parts[colMap.date].replace(/"/g, '').trim() : 'N/A', sku: parts[colMap.sku] ? parts[colMap.sku].replace(/"/g, '').trim() : 'N/A', name: parts[colMap.name] ? parts[colMap.name].replace(/"/g, '') : 'N/A', price: paidPrice, commission: commission > 0 ? commission : 0, logistics: colMap.logistics !== -1 ? cleanNumber(parts[colMap.logistics]) : 0, quantity: cleanNumber(parts[colMap.quantity]) }); } console.log(`Успешно загружено строк: \${ozonData.length}`); } // Вспомогательная функция для правильного разбиения CSV (учитывает кавычки) function parseCSVLine(line, delimiter) { const result = []; let current = ''; let inQuotes = false; for (let i = 0; i < line.length; i++) { const char = line[i]; const nextChar = line[i + 1]; if (char === '"') { if (inQuotes && nextChar === '"') { current += '"'; // Экранированная кавычка i++; } else { inQuotes = !inQuotes; } } else if (char === delimiter && !inQuotes) { result.push(current); current = ''; } else { current += char; } } result.push(current); return result; } // Вспомогательная функция для поиска индекса колонки function findColumnIndex(headers, possibleNames) { for (const name of possibleNames) { const index = headers.findIndex(h => h.includes(name.toLowerCase())); if (index !== -1) return index; } return -1; }
