// В функции parseOzonCSV добавим явное указание колонок function parseOzonCSV(text) { ozonData = []; const lines = text.split('\n'); if (lines.length < 2) return; const firstLine = lines[0]; const delimiter = firstLine.includes(';') ? ';' : ','; const headers = firstLine.split(delimiter).map(h => h.replace(/"/g, '').trim().toLowerCase()); // Явно указываем позиции колонок const colMap = { date: findColumn(headers, ['дата отгрузки']), sku: findColumn(headers, ['sku']), name: findColumn(headers, ['название товара']), price: findColumn(headers, ['оплачено покупателем']), // Используем оплаченную сумму commission: findColumn(headers, ['стоимость доставки']), quantity: findColumn(headers, ['количество']) }; // Добавляем расчет комиссии Ozon // В вашем файле нет прямой комиссии, но её можно рассчитать как разницу // между "Ваша цена" и "Оплачено покупателем" for (let i = 1; i < lines.length; i++) { const line = lines[i].trim(); if (!line) continue; const parts = line.split(delimiter); if (parts.length <= colMap.price) continue; const clean = (val) => parseFloat(val.replace(/"/g, '').replace(/ /g, '').replace(',', '.')); ozonData.push({ date: parts[colMap.date] ? parts[colMap.date].replace(/"/g, '') : 'N/A', sku: parts[colMap.sku] ? parts[colMap.sku].replace(/"/g, '').trim() : 'N/A', name: parts[colMap.name] ? parts[colMap.name].replace(/"/g, '') : 'N/A', price: clean(parts[colMap.price]), commission: clean(parts[findColumn(headers, ['ваша цена'])] - parts[colMap.price]), // Расчет комиссии logistics: clean(parts[colMap.commission]), quantity: clean(parts[colMap.quantity]) }); } }
